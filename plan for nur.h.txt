আমি আপনার জন্য একটি production-ready `nur.h` ফাইল তৈরি করছি যা ISO C স্ট্যান্ডার্ড এবং C23 উভয়ের সাথে সামঞ্জস্যপূর্ণ।এখন আমি একটি সম্পূর্ণ production-ready `nur.h` ফাইল তৈরি করব:

```c
/*
 * nur.h - Safe Error Handling for C
 * 
 * A modern, Rust-inspired Result<T,E> error handling system for C.
 * Provides typed outcomes with explicit error propagation and 
 * structured diagnostics.
 * 
 * Features:
 * - NurResult<T> - Tagged union for success/error outcomes
 * - NurOption<T> - Optional value abstraction  
 * - NurStatus - Void operations with error states
 * - Structured error objects with source location
 * - Rust-style error propagation (NUR_TRY)
 * - C11/C17/C23 compatibility with optional modern features
 * - Zero-allocation core with optional formatting
 * 
 * License: MIT
 * Version: 1.0.0
 */

#ifndef NUR_H
#define NUR_H

/* ========================================================================== */
/*                            VERSION & CONFIG                               */
/* ========================================================================== */

#define NUR_VERSION_MAJOR 1
#define NUR_VERSION_MINOR 0  
#define NUR_VERSION_PATCH 0
#define NUR_VERSION_STRING "1.0.0"

/* Feature detection */
#if defined(__STDC_VERSION__)
    #if __STDC_VERSION__ >= 202311L
        #define NUR_C23 1
    #elif __STDC_VERSION__ >= 201710L  
        #define NUR_C17 1
    #elif __STDC_VERSION__ >= 201112L
        #define NUR_C11 1
    #elif __STDC_VERSION__ >= 199901L
        #define NUR_C99 1
    #endif
#endif

/* Compiler extensions */
#if defined(__GNUC__) || defined(__clang__)
    #define NUR_GNU_EXTENSIONS 1
#endif

/* ========================================================================== */
/*                              INCLUDES                                     */
/* ========================================================================== */

#include <stdbool.h>
#include <stddef.h>

/* Optional includes based on configuration */
#ifndef NUR_NO_ERRNO
    #include <errno.h>
#endif

#ifndef NUR_NO_STRING

#endif

/* [[nodiscard]] equivalent for different C versions */
#ifdef NUR_C23
    #define NUR_NODISCARD [[nodiscard]]
    #define NUR_NODISCARD_MSG(msg) [[nodiscard(msg)]]
#elif defined(NUR_GNU_EXTENSIONS)
    #define NUR_NODISCARD __attribute__((warn_unused_result))
    #define NUR_NODISCARD_MSG(msg) __attribute__((warn_unused_result))
#elif defined(_MSC_VER) && _MSC_VER >= 1700
    #define NUR_NODISCARD _Check_return_
    #define NUR_NODISCARD_MSG(msg) _Check_return_
#else
    #define NUR_NODISCARD
    #define NUR_NODISCARD_MSG(msg)
#endif

/* Function attributes */
#ifdef NUR_GNU_EXTENSIONS
    #define NUR_INLINE static inline __attribute__((always_inline))
    #define NUR_LIKELY(x) __builtin_expect(!!(x), 1)
    #define NUR_UNLIKELY(x) __builtin_expect(!!(x), 0)
#else
    #define NUR_INLINE static inline
    #define NUR_LIKELY(x) (x)
    #define NUR_UNLIKELY(x) (x)
#endif

/* Core error codes */
typedef enum {
    NUR_E_NONE = 0,
    
    /* Input/validation errors */
    NUR_E_INVALID_ARGUMENT = 1,
    NUR_E_OUT_OF_RANGE = 2,
    NUR_E_NULL_POINTER = 3,
    NUR_E_INVALID_STATE = 4,
    
    /* Data errors */
    NUR_E_PARSE = 10,
    NUR_E_FORMAT = 11, 
    NUR_E_ENCODING = 12,
    NUR_E_CORRUPTED = 13,
    
    /* Resource errors */
    NUR_E_OUT_OF_MEMORY = 20,
    NUR_E_RESOURCE_LIMIT = 21,
    NUR_E_BUSY = 22,
    
    /* I/O errors */
    NUR_E_NOT_FOUND = 30,
    NUR_E_PERMISSION = 31,
    NUR_E_ALREADY_EXISTS = 32,
    NUR_E_IO = 33,
    
    /* Network/async errors */
    NUR_E_TIMEOUT = 40,
    NUR_E_CANCELLED = 41,
    NUR_E_CONNECTION = 42,
    NUR_E_PROTOCOL = 43,
    
    /* System errors */
    NUR_E_UNSUPPORTED = 50,
    NUR_E_INTERNAL = 51,
    NUR_E_BUG = 52,
    
    /* User-defined errors start at 1000 */
    NUR_E_USER_BASE = 1000
} nur_error_code_t;

/* Rich error object */
typedef struct {
    nur_error_code_t code;
    int system_code;        /* errno, win32 error, etc. */
    const char *message;    /* Human-readable description */
    const char *context;    /* Operation context */
    
    /* Source location (optional) */
#ifndef NUR_NO_SOURCE_LOCATION
    const char *file;
    const char *function;
    unsigned line;
#endif
} nur_error_t;

/* ========================================================================== */
/*                          ERROR CONSTRUCTORS                               */
/* ========================================================================== */

NUR_INLINE nur_error_t nur_error_make(
    nur_error_code_t code,
    const char *message,
    const char *context
#ifndef NUR_NO_SOURCE_LOCATION    
    , const char *file,
    const char *function,
    unsigned line
#endif
) {
    nur_error_t err = {0};
    err.code = code;
    err.system_code = 0;
    err.message = message ? message : "unknown error";
    err.context = context;
    
#ifndef NUR_NO_SOURCE_LOCATION
    err.file = file;
    err.function = function; 
    err.line = line;
#endif
    
    return err;
}

#ifndef NUR_NO_SOURCE_LOCATION
    #define NUR_ERROR(code, msg) \
        nur_error_make((code), (msg), NULL, __FILE__, __func__, __LINE__)
        
    #define NUR_ERROR_CTX(code, msg, ctx) \
        nur_error_make((code), (msg), (ctx), __FILE__, __func__, __LINE__)
#else
    #define NUR_ERROR(code, msg) \
        nur_error_make((code), (msg), NULL)
        
    #define NUR_ERROR_CTX(code, msg, ctx) \
        nur_error_make((code), (msg), (ctx))
#endif

/* errno integration */
#ifndef NUR_NO_ERRNO
NUR_INLINE nur_error_t nur_error_from_errno(
    const char *message,
    const char *context
#ifndef NUR_NO_SOURCE_LOCATION
    , const char *file,
    const char *function,  
    unsigned line
#endif
) {
    nur_error_t err = {0};
    err.code = NUR_E_IO; /* Default mapping */
    err.system_code = errno;
    
#ifndef NUR_NO_STRING
    err.message = message ? message : strerror(errno);
#else
    err.message = message ? message : "system error";
#endif
    
    err.context = context;
    
#ifndef NUR_NO_SOURCE_LOCATION
    err.file = file;
    err.function = function;
    err.line = line;
#endif
    
    return err;
}

#ifndef NUR_NO_SOURCE_LOCATION
    #define NUR_ERROR_ERRNO(msg) \
        nur_error_from_errno((msg), NULL, __FILE__, __func__, __LINE__)
        
    #define NUR_ERROR_ERRNO_CTX(msg, ctx) \
        nur_error_from_errno((msg), (ctx), __FILE__, __func__, __LINE__)
#else
    #define NUR_ERROR_ERRNO(msg) \
        nur_error_from_errno((msg), NULL)
        
    #define NUR_ERROR_ERRNO_CTX(msg, ctx) \
        nur_error_from_errno((msg), (ctx))
#endif

#endif /* NUR_NO_ERRNO */

/* Result type generator macro */
#define NUR_RESULT_TYPE(T, Name) \
    typedef struct NUR_NODISCARD_MSG("Handle the NurResult") { \
        bool ok; \
        union { \
            T value; \
            nur_error_t error; \
        } u; \
    } NurResult_##Name; \
    \
    NUR_INLINE NurResult_##Name nur_ok_##Name(T val) { \
        NurResult_##Name r = {0}; \
        r.ok = true; \
        r.u.value = val; \
        return r; \
    } \
    \
    NUR_INLINE NurResult_##Name nur_err_##Name(nur_error_t err) { \
        NurResult_##Name r = {0}; \
        r.ok = false; \
        r.u.error = err; \
        return r; \
    }

/* Predefined common result types */
NUR_RESULT_TYPE(int, i32)
NUR_RESULT_TYPE(long, i64)  
NUR_RESULT_TYPE(size_t, usize)
NUR_RESULT_TYPE(float, f32)
NUR_RESULT_TYPE(double, f64)
NUR_RESULT_TYPE(char *, cstr)
NUR_RESULT_TYPE(void *, ptr)

/* Void result (status-only) */
typedef struct NUR_NODISCARD_MSG("Handle the status") {
    bool ok;
    nur_error_t error; /* Only valid when ok == false */
} NurResult_void;

NUR_INLINE NurResult_void nur_ok_void(void) {
    NurResult_void r = {0};
    r.ok = true;
    return r;
}

NUR_INLINE NurResult_void nur_err_void(nur_error_t err) {
    NurResult_void r = {0}; 
    r.ok = false;
    r.error = err;
    return r;
}

    typedef struct { \
        bool some; \
        T value; /* Only valid when some == true */ \
    } NurOption_##Name; \
    \
    NUR_INLINE NurOption_##Name nur_some_##Name(T val) { \

        o.some = true; \

    } \
    \

NUR_OPTION_TYPE(int, i32)

/* Construction shortcuts */
#define NUR_OK(Name, val) nur_ok_##Name(val)
#define NUR_ERR(Name, err) nur_err_##Name(err)
#define NUR_OK_VOID() nur_ok_void()
#define NUR_ERR_VOID(err) nur_err_void(err)

#define NUR_SOME(Name, val) nur_some_##Name(val)
#define NUR_NONE(Name) nur_none_##Name()

/* Inspection */
#define NUR_IS_OK(r) ((r).ok)
#define NUR_IS_ERR(r) (!(r).ok)
#define NUR_IS_SOME(o) ((o).some)
#define NUR_IS_NONE(o) (!(o).some)

/* Unsafe value extraction (use with care) */
#define NUR_VALUE_UNSAFE(r) ((r).u.value)
#define NUR_ERROR_UNSAFE(r) ((r).u.error)
#define NUR_OPTION_VALUE_UNSAFE(o) ((o).value)

/* Safe value extraction with panic */
#ifndef NUR_NO_STDIO
    #define NUR_UNWRAP(r) \
        (NUR_LIKELY((r).ok) ? (r).u.value : \
         (fprintf(stderr, "[NUR PANIC] Unwrapped error at %s:%d: %s\n", \
                  __FILE__, __LINE__, (r).u.error.message), \
          exit(1), (r).u.value))
          
    #define NUR_EXPECT(r, msg) \
        (NUR_LIKELY((r).ok) ? (r).u.value : \
         (fprintf(stderr, "[NUR PANIC] %s at %s:%d: %s\n", \
                  (msg), __FILE__, __LINE__, (r).u.error.message), \
          exit(1), (r).u.value))
#endif

/* Value with fallback */  
#define NUR_UNWRAP_OR(r, fallback) \
    (NUR_LIKELY((r).ok) ? (r).u.value : (fallback))

#define NUR_OPTION_UNWRAP_OR(o, fallback) \
    (NUR_LIKELY((o).some) ? (o).value : (fallback))

/* ========================================================================== */
/*                         ERROR PROPAGATION                                 */
/* ========================================================================== */

/* Modern C with __auto_type (GCC/Clang extension) */
#ifdef NUR_GNU_EXTENSIONS
    #define NUR_TRY(expr) __extension__ ({ \
        __auto_type _nur_r = (expr); \
        if (NUR_UNLIKELY(!_nur_r.ok)) \
            return nur_err_void(_nur_r.u.error); \
        _nur_r.u.value; \
    })
    
    #define NUR_TRY_AS(DestType, expr) __extension__ ({ \
        __auto_type _nur_r = (expr); \
        if (NUR_UNLIKELY(!_nur_r.ok)) \
            return nur_err_##DestType(_nur_r.u.error); \
        _nur_r.u.value; \
    })
#endif

/* Portable C99+ version */
#define NUR_TRY_TO(DestType, SrcType, lhs, expr) \
    do { \
        NurResult_##SrcType _nur_r = (expr); \
        if (NUR_UNLIKELY(!_nur_r.ok)) \
            return nur_err_##DestType(_nur_r.u.error); \
        (lhs) = _nur_r.u.value; \
    } while (0)

/* Check condition and return error if false */
#define NUR_CHECK(cond, err) \
    do { \
        if (NUR_UNLIKELY(!(cond))) \
            return nur_err_void(err); \
    } while (0)

#define NUR_CHECK_AS(DestType, cond, err) \
    do { \
        if (NUR_UNLIKELY(!(cond))) \
            return nur_err_##DestType(err); \
    } while (0)

/* Explicitly ignore a result (document intent) */
#define NUR_IGNORE(expr) ((void)(expr))

#ifndef NUR_NO_STDIO
NUR_INLINE void nur_error_print(const nur_error_t *err, FILE *f) {
    if (!err || !f) return;
    
    fprintf(f, "[%s]", nur_error_code_name(err->code));
    
    if (err->message) {
        fprintf(f, " %s", err->message);
    }
    
    if (err->context) {
        fprintf(f, " (context: %s)", err->context);
    }
    
    if (err->system_code != 0) {
        fprintf(f, " (system: %d)", err->system_code);
    }
    
#ifndef NUR_NO_SOURCE_LOCATION
    if (err->file && err->function) {
        fprintf(f, " (at %s:%u in %s())", err->file, err->line, err->function);
    }
#endif
    
    fprintf(f, "\n");
}

#define nur_error_eprint(err) nur_error_print((err), stderr)

#ifndef NUR_NO_ERRNO
/* POSIX-style function adapter */
NUR_INLINE NurResult_i32 nur_open(const char *path, int flags) {
    if (!path) {
        return NUR_ERR(i32, NUR_ERROR(NUR_E_NULL_POINTER, "path is null"));
    }
    
    int fd = open(path, flags);
    if (fd < 0) {
        return NUR_ERR(i32, NUR_ERROR_ERRNO_CTX("open failed", path));
    }
    
    return NUR_OK(i32, fd);
}

/* malloc wrapper */
NUR_INLINE NurResult_ptr nur_malloc(size_t size) {
    if (size == 0) {
        return NUR_ERR(ptr, NUR_ERROR(NUR_E_INVALID_ARGUMENT, "zero size"));
    }
    
    void *p = malloc(size);
    if (!p) {
        return NUR_ERR(ptr, NUR_ERROR(NUR_E_OUT_OF_MEMORY, "allocation failed"));
    }
    
    return NUR_OK(ptr, p);
}

/*
 * Usage Guidelines:
 * 
 * 1. Always check .ok before accessing .u.value
 * 2. Use NUR_TRY for error propagation in modern compilers  
 * 3. Use NUR_TRY_TO for portable error propagation
 * 4. Mark fallible functions with NUR_NODISCARD where supported
 * 5. Use NUR_IGNORE() to explicitly document ignored results
 * 6. Prefer error codes over string matching for control flow
 * 7. Keep error messages descriptive but not excessive
 * 8. Use context for operation-specific information
 * 
 * Thread Safety:
 * - Result objects are value types and naturally thread-safe
 * - Error objects with static message pointers are safe
 * - errno capture should happen immediately after system calls
 * 
 * Memory Management:
 * - Core nur.h does not allocate memory
 * - Message/context pointers should be static or caller-managed  
 * - Large payloads should be passed by pointer in Result types
 * 
 * Compatibility:
 * - C99: Basic functionality, manual error propagation
 * - C11: All features with compiler extensions
 * - C23: Native [[nodiscard]] support
```

এখন একটি সহজ example ফাইল তৈরি করি যা দেখাবে কীভাবে এই library ব্যবহার করতে হয়:

```c
/*
 * nur_example.c - Usage examples for nur.h
 * 
 * Compile with:
 *   gcc -std=c99 -Wall -Wextra -o example nur_example.c
 *   gcc -std=c11 -Wall -Wextra -o example nur_example.c  (recommended)
 *   gcc -std=c23 -Wall -Wextra -o example nur_example.c  (latest)
 */

#include "nur.h"
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

/* Example 1: Basic integer parsing with detailed error handling */
NurResult_i32 parse_int(const char *str) {
    if (!str) {
        return NUR_ERR(i32, NUR_ERROR(NUR_E_NULL_POINTER, "input string is null"));
    }
    
    if (*str == '\0') {
        return NUR_ERR(i32, NUR_ERROR(NUR_E_INVALID_ARGUMENT, "empty string"));
    }
    
    errno = 0;
    char *endptr;
    long val = strtol(str, &endptr, 10);
    
    if (endptr == str) {
        return NUR_ERR(i32, NUR_ERROR_CTX(NUR_E_PARSE, "no valid digits found", str));
    }
    
    if (*endptr != '\0') {
        return NUR_ERR(i32, NUR_ERROR_CTX(NUR_E_PARSE, "trailing characters", str));
    }
    
    if (errno == ERANGE || val < INT_MIN || val > INT_MAX) {
        return NUR_ERR(i32, NUR_ERROR_CTX(NUR_E_OUT_OF_RANGE, "integer overflow", str));
    }
    
    return NUR_OK(i32, (int)val);
}

/* Example 2: Division with error propagation */
NurResult_i32 safe_divide(int a, int b) {
    if (b == 0) {
        return NUR_ERR(i32, NUR_ERROR(NUR_E_INVALID_ARGUMENT, "division by zero"));
    }
    
    return NUR_OK(i32, a / b);
}

/* Example 3: Error propagation using NUR_TRY_TO (portable) */  
NurResult_i32 parse_and_divide(const char *a_str, const char *b_str) {
    int a, b;
    
    /* Parse first number */
    NUR_TRY_TO(i32, i32, a, parse_int(a_str));
    
    /* Parse second number */ 
    NUR_TRY_TO(i32, i32, b, parse_int(b_str));
    
    /* Perform division */
    return safe_divide(a, b);
}

/* Example 4: Option type usage */
NurOption_i32 find_in_array(const int *arr, size_t len, int target) {
    if (!arr) {
        return NUR_NONE(i32);
    }
    
    for (size_t i = 0; i < len; i++) {
        if (arr[i] == target) {
            return NUR_SOME(i32, (int)i);
        }
    }
    
    return NUR_NONE(i32);
}

/* Example 5: File operations with cleanup */
NurResult_cstr read_file_content(const char *filename) {
    if (!filename) {
        return NUR_ERR(cstr, NUR_ERROR(NUR_E_NULL_POINTER, "filename is null"));
    }
    
    FILE *file = fopen(filename, "r");
    if (!file) {
        return NUR_ERR(cstr, NUR_ERROR_ERRNO_CTX("failed to open file", filename));
    }
    
    /* Get file size */
    if (fseek(file, 0, SEEK_END) != 0) {
        fclose(file);
        return NUR_ERR(cstr, NUR_ERROR_ERRNO("fseek failed"));
    }
    
    long size = ftell(file);
    if (size < 0) {
        fclose(file);
        return NUR_ERR(cstr, NUR_ERROR_ERRNO("ftell failed"));
    }
    
    rewind(file);
    
    /* Allocate buffer */
    char *buffer = malloc((size_t)size + 1);
    if (!buffer) {
        fclose(file);
        return NUR_ERR(cstr, NUR_ERROR(NUR_E_OUT_OF_MEMORY, "failed to allocate buffer"));
    }
    
    /* Read content */
    size_t read_size = fread(buffer, 1, (size_t)size, file);
    fclose(file);
    
    if (read_size != (size_t)size) {
        free(buffer);
        return NUR_ERR(cstr, NUR_ERROR(NUR_E_IO, "incomplete file read"));
    }
    
    buffer[size] = '\0';
    return NUR_OK(cstr, buffer);
}

/* Demonstration function */
void demonstrate_nur_usage(void) {
    printf("=== Nur.h Safe Error Handling Demo ===\n\n");
    
    /* Example 1: Basic parsing */
    printf("1. Integer parsing:\n");
    const char *test_strings[] = {"42", "-123", "abc", "123abc", "", NULL};
    
    for (size_t i = 0; i < sizeof(test_strings)/sizeof(test_strings[0]); i++) {
        const char *str = test_strings[i];
        NurResult_i32 result = parse_int(str);
        
        printf("   parse_int(\"%s\") = ", str ? str : "(null)");
        
        if (NUR_IS_OK(result)) {
            printf("OK(%d)\n", NUR_VALUE_UNSAFE(result));
        } else {
            printf("ERR(");
            nur_error_print(&NUR_ERROR_UNSAFE(result), stdout);
            printf("   )\n");
        }
    }
    
    /* Example 2: Error propagation */
    printf("\n2. Error propagation in calculations:\n");
    NurResult_i32 calc_result = parse_and_divide("10", "2");
    
    if (NUR_IS_OK(calc_result)) {
        printf("   10/2 = %d\n", NUR_VALUE_UNSAFE(calc_result));
    } else {
        printf("   Calculation failed: ");
        nur_error_eprint(&NUR_ERROR_UNSAFE(calc_result));
    }
    
    calc_result = parse_and_divide("10", "0");
    if (NUR_IS_ERR(calc_result)) {
        printf("   Division by zero caught: ");
        nur_error_eprint(&NUR_ERROR_UNSAFE(calc_result));
    }
    
    /* Example 3: Option type */
    printf("\n3. Option type usage:\n");
    int arr[] = {1, 3, 5, 7, 9};
    size_t arr_len = sizeof(arr) / sizeof(arr[0]);
    
    NurOption_i32 found = find_in_array(arr, arr_len, 5);
    if (NUR_IS_SOME(found)) {
        printf("   Found 5 at index %d\n", NUR_OPTION_VALUE_UNSAFE(found));
    }
    
    found = find_in_array(arr, arr_len, 4);
    if (NUR_IS_NONE(found)) {
        printf("   4 not found in array\n");
    }
    
    /* Example 4: Unwrap with fallback */
    printf("\n4. Safe value extraction:\n");
    NurResult_i32 good = NUR_OK(i32, 42);
    NurResult_i32 bad = NUR_ERR(i32, NUR_ERROR(NUR_E_INVALID_ARGUMENT, "test error"));
    
    printf("   Good result value: %d\n", NUR_UNWRAP_OR(good, -1));
    printf("   Bad result fallback: %d\n", NUR_UNWRAP_OR(bad, -1));
    
    printf("\n=== Demo completed ===\n");
}
```
