🚀 nur.h

Rust-style Result<T,E> Error Handling for C

Production-Ready • Zero-Allocation • Multi-Standard • Cross-Platform

nur.h কী এবং কেন ব্যবহার করবেন

nur.h হলো একটি সিঙ্গেল-ফাইল, হেডার-অনলি C লাইব্রেরি যা Rust-স্টাইলের নিরাপদ error handling আপনার C প্রোগ্রামে যোগ করে। এটি FAANG/MANGO স্ট্যান্ডার্ডে তৈরি একটি প্রোডাকশন-রেডি লাইব্রেরি।

কোর লাইব্রেরিতে কোনো malloc নেই - সম্পূর্ণ স্ট্যাক-ভিত্তিক

🔒 টাইপ-সেফ

C99, C11, C17, C23 সব স্ট্যান্ডার্ডে কাজ করে

  • 50+ Error Codes: সুসংগঠিত error taxonomy

🎓 সেরা অনুশীলন (Best Practices)

✅ করণীয়:

  • সবসময় Result চেক করুন NUR_IS_OK() দিয়ে
  • Error message স্পষ্ট রাখুন
  • NUR_ERROR_CTX() ব্যবহার করে context যোগ করুন
  • Custom types তৈরি করুন NUR_RESULT_TYPE() দিয়ে

❌ করণীয় নয়:

  • Result ignore করবেন না
  • Error checked না করে value access করবেন না
  • NULL pointer pass করবেন না
  • Memory leak করবেন না

কোড উদাহরণসমূহ

উদাহরণ ১: সাধারণ Error Handling

#include <nur.h>
#include <stdio.h>

NurResult_i32 divide(int a, int b) {
    if (b == 0) {
        return NUR_ERR(i32, NUR_ERROR(NUR_E_INVALID_ARGUMENT, 
                                       "ভাগ শূন্য দিয়ে করা যায় না"));
    }
    return NUR_OK(i32, a / b);
}

int main(void) {
    NurResult_i32 result = divide(10, 2);
    
    if (NUR_IS_OK(result)) {
        printf("ফলাফল: %d\n", NUR_VALUE(result));
    } else {
        nur_error_t err = NUR_ERROR_OF(result);
        printf("এরর: %s\n", err.message);
    }
    
    return 0;
}

উদাহরণ ২: File I/O Error Handling

#include <nur.h>
#include <stdio.h>

NurResult_ptr read_file(const char *filename) {
    FILE *file = fopen(filename, "r");
    if (file == NULL) {
        return NUR_ERR(ptr, NUR_ERROR_ERRNO("ফাইল খুলতে ব্যর্থ"));
    }
    
    fseek(file, 0, SEEK_END);
    long size = ftell(file);
    rewind(file);
    
    NurResult_ptr mem = nur_malloc(size + 1);
    if (NUR_IS_ERR(mem)) {
        fclose(file);
        return mem;
    }
    
    char *buffer = (char *)NUR_VALUE(mem);
    fread(buffer, 1, size, file);
    buffer[size] = '\0';
    fclose(file);
    
    return NUR_OK(ptr, buffer);
}

int main(void) {
    NurResult_ptr result = read_file("data.txt");
    
    if (NUR_IS_OK(result)) {
        char *content = (char *)NUR_VALUE(result);
        printf("Content: %s\n", content);
        free(content);
    } else {
        nur_error_eprint(&NUR_ERROR_OF(result));
    }
    
    return 0;
}

উদাহরণ ৩: Custom Type Result

#include <nur.h>
#include <stdio.h>
#include <string.h>

typedef struct {
    char name[50];
    int age;
} Person;

// Person এর জন্য Result type তৈরি
NUR_RESULT_TYPE(Person, person)

NurResult_person create_person(const char *name, int age) {
    if (name == NULL) {
        return NUR_ERR(person, NUR_ERROR(NUR_E_NULL_POINTER, 
                                         "নাম NULL হতে পারে না"));
    }
    
    if (age < 0 || age > 150) {
        return NUR_ERR(person, NUR_ERROR(NUR_E_OUT_OF_RANGE, 
                                         "বয়স অবৈধ"));
    }
    
    Person p;
    strncpy(p.name, name, sizeof(p.name) - 1);
    p.name[sizeof(p.name) - 1] = '\0';
    p.age = age;
    
    return NUR_OK(person, p);
}

int main(void) {
    NurResult_person result = create_person("রহিম", 30);
    
    if (NUR_IS_OK(result)) {
        Person p = NUR_VALUE(result);
        printf("নাম: %s, বয়স: %d\n", p.name, p.age);
    } else {
        nur_error_eprint(&NUR_ERROR_OF(result));
    }
    
    return 0;
}

কাজ ম্যাক্রো/ফাংশন
Success তৈরি NUR_OK(suffix, value)
Error তৈরি NUR_ERROR(code, msg)
Success চেক NUR_IS_OK(result)
Error চেক NUR_IS_ERR(result)
Value নেওয়া NUR_VALUE(result)
Error propagate NUR_TRY_TO(Dest, Src, lhs, expr)
